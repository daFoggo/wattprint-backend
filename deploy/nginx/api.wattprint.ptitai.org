# FastAPI for the mobile app (127.0.0.1:11093). The web dashboard calls it server-side and does not need this site.
# Every route but /health and /homes/provinces needs a Keycloak bearer token. Installed by ./apply.sh.
server {
    listen 80;
    server_name api.wattprint.ptitai.org;

    location ^~ /.well-known/acme-challenge/ {
        root /var/www/letsencrypt;
        default_type text/plain;
    }

    location / {
        return 301 https://$host$request_uri;
    }
}

# --- 443 (installed once the certificate exists) ---
server {
    listen 443 ssl;
    server_name api.wattprint.ptitai.org;
    ssl_certificate /etc/letsencrypt/live/api.wattprint.ptitai.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/api.wattprint.ptitai.org/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
    client_max_body_size 25m;  # voice clips (POST /api/v1/copilot/voice/transcribe)

    # The app does not use the interactive docs; keep the API map private.
    location ~ ^/(docs|redoc|openapi\.json)(/|$) { return 404; }

    location /api/ {
        proxy_pass http://127.0.0.1:11093;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $remote_addr;  # the edge: drop what the client sent
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Connection "";
        # SSE: /api/v1/energy/stream and the copilot chat stream must not sit in a buffer.
        proxy_buffering off;
        proxy_request_buffering off;
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;
    }

    location / { return 404; }
}
