# Landing + dashboard (multi-zone: apps/web rewrites the dashboard paths to apps/dashboard).
# Upstream is the prod worktree (`next start -H localhost -p 11081`), not the dev servers. Installed by ./apply.sh.
server {
    listen 80;
    server_name wattprint.ptitai.org;

    location ^~ /.well-known/acme-challenge/ {
        root /var/www/letsencrypt;
        default_type text/plain;
    }

    location / {
        return 301 https://$host$request_uri;
    }
}

# --- 443 (installed once the certificate exists) ---
server {
    listen 443 ssl;
    server_name wattprint.ptitai.org;
    ssl_certificate /etc/letsencrypt/live/wattprint.ptitai.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/wattprint.ptitai.org/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
    client_max_body_size 25m;  # Wappi voice clips (POST /api/copilot/voice/transcribe)
    # /auth/callback sets the access, refresh and id tokens as cookies: past the 4k default nginx answers 502.
    proxy_buffer_size 128k;
    proxy_buffers 4 256k;
    proxy_busy_buffers_size 256k;

    # Wappi's routes exist only in apps/dashboard and apps/web does not rewrite /api: send them straight there
    # (also keeps the chat stream off Next's 30 s rewrite-proxy timeout).
    location ^~ /api/copilot/ {
        proxy_pass http://127.0.0.1:11082;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $remote_addr;  # the edge: drop what the client sent
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Connection "";
        proxy_buffering off;
        proxy_request_buffering off;
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;
    }

    location / {
        proxy_pass http://127.0.0.1:11081;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $remote_addr;  # the edge: drop what the client sent
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header Connection "";
        # Streamed RSC responses must not sit in a buffer.
        proxy_buffering off;
        proxy_request_buffering off;
        proxy_redirect off;
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;
    }

    # Security headers (CSP, nosniff, Referrer-Policy) come from the Next.js apps; repeating them here duplicates them.
}
