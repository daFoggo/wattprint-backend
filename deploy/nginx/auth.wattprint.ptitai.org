# Keycloak (realm wattprint) at 127.0.0.1:11095; compose sets KC_PROXY_HEADERS=xforwarded. Installed by ./apply.sh.
server {
    listen 80;
    server_name auth.wattprint.ptitai.org;

    location ^~ /.well-known/acme-challenge/ {
        root /var/www/letsencrypt;
        default_type text/plain;
    }

    location / {
        return 301 https://$host$request_uri;
    }
}

# --- 443 (installed once the certificate exists) ---
server {
    listen 443 ssl;
    server_name auth.wattprint.ptitai.org;
    ssl_certificate /etc/letsencrypt/live/auth.wattprint.ptitai.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/auth.wattprint.ptitai.org/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # The admin console and the master realm stay private: reach them through localhost:11095 (VS Code port forward).
    location ^~ /admin { return 404; }
    location ^~ /realms/master { return 404; }

    location / {
        proxy_pass http://127.0.0.1:11095;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $remote_addr;  # the edge: drop what the client sent
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Port 443;
        # Keycloak's auth responses carry large cookies and headers.
        proxy_buffer_size 128k;
        proxy_buffers 4 256k;
        proxy_busy_buffers_size 256k;
    }
}
